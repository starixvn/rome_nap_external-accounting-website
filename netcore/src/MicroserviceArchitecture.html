1.2. Nguyên tắc thiết kế Microservice

  • Single Responsibility: Mỗi service có một trách nhiệm duy nhất
  • Database per Service: Mỗi service có database riêng
  • Decentralized Governance: Tự quản lý và phát triển
  • Fault Tolerance: Khả năng chịu lỗi cao
  • Observability: Monitoring và logging toàn diện

2. API Gateway và Routing

2.1. API Gateway Configuration

// Program.cs - API Gateway var builder = WebApplication.CreateBuilder(args); // Add services builder.Services.AddControllers(); builder.Services.AddEndpointsApiExplorer(); builder.Services.AddSwaggerGen(); // Configure API Gateway builder.Services.AddReverseProxy() .LoadFromConfig(builder.Configuration.GetSection("ReverseProxy")); var app = builder.Build(); // Configure pipeline app.UseRouting(); app.UseAuthentication(); app.UseAuthorization(); // Map reverse proxy app.MapReverseProxy(); app.Run();

5. Inter-Service Communication

5.1. Synchronous Communication (HTTP/REST)

// HTTP Client for inter-service communication public class CustomerServiceClient { private readonly HttpClient _httpClient; public CustomerServiceClient(HttpClient httpClient) { _httpClient = httpClient; } public async Task GetCustomerAsync(int customerId) { var response = await _httpClient.GetAsync($"/api/customer/{customerId}"); response.EnsureSuccessStatusCode(); var content = await response.Content.ReadAsStringAsync(); return JsonSerializer.Deserialize(content); } }

5.2. Asynchronous Communication (Message Queue)

// Message Queue Configuration public class MessageQueueService { private readonly IMessageBus _messageBus; public async Task PublishCustomerCreatedEvent(CustomerCreatedEvent @event) { await _messageBus.PublishAsync(@event); } public async Task SubscribeToEvents() { await _messageBus.SubscribeAsync(HandleCustomerCreated); await _messageBus.SubscribeAsync(HandleReportRequested); } }

5.3. Service Discovery

// Service Discovery Configuration public class ServiceDiscoveryService { private readonly IConsulClient _consulClient; public async Task GetServiceUrl(string serviceName) { var services = await _consulClient.Health.Service(serviceName, "", true); var service = services.Response.FirstOrDefault(); return service != null ? $"http://{service.Service.Address}:{service.Service.Port}" : throw new ServiceNotFoundException(serviceName); } }

6. Security và Authentication

6.1. JWT Token Strategy

// JWT Configuration public class JwtService { public string GenerateToken(User user) { var claims = new[] { new Claim(ClaimTypes.NameIdentifier, user.Id.ToString()), new Claim(ClaimTypes.Name, user.Username), new Claim(ClaimTypes.Role, user.Role), new Claim("service", "auth-service") }; var key = new SymmetricSecurityKey(Encoding.UTF8.GetBytes(_jwtSettings.SecretKey)); var credentials = new SigningCredentials(key, SecurityAlgorithms.HmacSha256); var token = new JwtSecurityToken( issuer: _jwtSettings.Issuer, audience: _jwtSettings.Audience, claims: claims, expires: DateTime.UtcNow.AddMinutes(_jwtSettings.ExpirationMinutes), signingCredentials: credentials ); return new JwtSecurityTokenHandler().WriteToken(token); } }

6.2. API Gateway Authentication

// API Gateway Authentication Middleware public class AuthenticationMiddleware { public async Task InvokeAsync(HttpContext context, RequestDelegate next) { var token = context.Request.Headers["Authorization"] .FirstOrDefault()?.Split(" ").Last(); if (token != null) { var principal = ValidateToken(token); context.User = principal; } await next(context); } }

7. Monitoring và Observability

7.1. Distributed Tracing

// OpenTelemetry Configuration public class TelemetryConfiguration { public static void ConfigureTelemetry(WebApplicationBuilder builder) { builder.Services.AddOpenTelemetry() .WithTracing(tracing => { tracing.AddAspNetCoreInstrumentation() .AddHttpClientInstrumentation() .AddEntityFrameworkCoreInstrumentation() .AddJaegerExporter(); }) .WithMetrics(metrics => { metrics.AddAspNetCoreInstrumentation() .AddHttpClientInstrumentation() .AddPrometheusExporter(); }); } }

7.2. Health Checks

// Health Check Configuration public class HealthCheckService { public static void ConfigureHealthChecks(IServiceCollection services) { services.AddHealthChecks() .AddCheck("database") .AddCheck("external-service") .AddCheck("memory"); } } // Health Check Implementation public class DatabaseHealthCheck : IHealthCheck { public async Task CheckHealthAsync(HealthCheckContext context, CancellationToken cancellationToken = default) { try { // Check database connectivity var isHealthy = await CheckDatabaseConnection(); return isHealthy ? HealthCheckResult.Healthy("Database is accessible") : HealthCheckResult.Unhealthy("Database is not accessible"); } catch (Exception ex) { return HealthCheckResult.Unhealthy("Database check failed", ex); } } }

WORKDIR "/src/AuthService" RUN dotnet build "AuthService.csproj" -c Release -o /app/build FROM build AS publish RUN dotnet publish "AuthService.csproj" -c Release -o /app/publish FROM base AS final WORKDIR /app COPY --from=publish /app/publish . ENTRYPOINT ["dotnet", "AuthService.dll"]

10. Best Practices và Recommendations

10.1. Microservice Best Practices

  • API Versioning: Sử dụng versioning cho API compatibility
  • Circuit Breaker: Implement circuit breaker pattern
  • Retry Policy: Cấu hình retry policy cho external calls
  • Rate Limiting: Implement rate limiting cho API protection
  • Graceful Shutdown: Implement graceful shutdown cho services

10.2. Monitoring và Alerting

  • Metrics: Collect business và technical metrics
  • Logging: Structured logging với correlation IDs
  • Alerting: Set up alerts cho critical issues
  • Dashboard: Real-time monitoring dashboard

10.3. Security Considerations

  • Network Security: Service mesh với mTLS
  • Data Encryption: Encrypt data in transit và at rest
  • Access Control: Role-based access control
  • Audit Logging: Comprehensive audit trail

11. Kết luận

Lợi ích của Microservice Architecture

  • Scalability: Scale từng service độc lập
  • Maintainability: Dễ maintain và update
  • Technology Diversity: Sử dụng công nghệ phù hợp cho từng service
  • Fault Isolation: Lỗi không ảnh hưởng toàn bộ system
  • Team Autonomy: Teams có thể phát triển độc lập

Thách thức cần lưu ý

  • Complexity: Tăng complexity trong quản lý
  • Network Latency: Communication overhead
  • Data Consistency: Distributed data management
  • Testing: Integration testing phức tạp hơn
  • Monitoring: Cần monitoring tools mạnh mẽ
